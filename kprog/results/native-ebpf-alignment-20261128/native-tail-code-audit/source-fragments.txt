1454: /// the native frame: the tail-call count is at [x26] (x26 = frame record - 16;
1455: /// P is built with -ffixed-x26), the frame's five callee-saved pairs lie below
1456: /// it. A taken tail call leaves P's own frames behind; a refused one falls through.
1457: fn build_arm64_tail_call_inline(helper_addrs: &HashMap<String, u64>) -> Result<Vec<u32>> {
1458:     let key = |k: &str| -> Result<u32> {
1459:         let v = *helper_addrs.get(k).ok_or_else(|| anyhow!("arm64 bpf_tail_call needs {k}"))?;
1460:         u32::try_from(v).map_err(|_| anyhow!("{k} does not fit u32"))
1461:     };
1462:     let max_entries = key(ARM64_BPF_MAP_MAX_ENTRIES_OFFSET_KEY)?;
1463:     let ptrs = key(ARM64_BPF_ARRAY_PTRS_OFFSET_KEY)?;
1464:     let bpf_func = key(ARM64_BPF_PROG_BPF_FUNC_OFFSET_KEY)?;
1465:     let entry = key(ARM64_TAIL_CALL_OFFSET_KEY)?;
1466:     if max_entries % 4 != 0 || max_entries / 4 > 0xfff || ptrs > 0xfff
1467:         || bpf_func % 8 != 0 || bpf_func / 8 > 0xfff || entry > 0xfff {
1468:         bail!("arm64 bpf_tail_call offsets exceed their encodings");
1469:     }
1470:     let add_imm = |rd: u32, rn: u32, imm: u32| 0x9100_0000 | (imm << 10) | (rn << 5) | rd;
1471:     let mut w = vec![
1472:         0x2a02_03e2,                                             // mov w2, w2
1473:         0xb940_0000 | ((max_entries / 4) << 10) | (1 << 5) | 16, // ldr w16, [x1, #max_entries]
1474:         0x6b10_005f,                                             // cmp w2, w16
1475:         0,                                                       // b.hs out
1476:         0xf940_0350,                                             // ldr x16, [x26]
1477:         0xf100_861f,                                             // cmp x16, #33 (MAX_TAIL_CALL_CNT)
1478:         0,                                                       // b.hs out
1479:         add_imm(15, 1, ptrs),                                    // add x15, x1, #ptrs
1480:         0xf862_79ee,                                             // ldr x14, [x15, x2, lsl #3]
1481:         0,                                                       // cbz x14, out
1482:         0x9100_0610,                                             // add x16, x16, #1
1483:         0xf900_0350,                                             // str x16, [x26]
1484:         0xd101_435f,                                             // sub sp, x26, #80
1485:         add_imm(29, 26, 16),                                     // add x29, x26, #16
1486:         0xa8c1_73fb, 0xa8c1_6bf9, 0xa8c1_63f7, 0xa8c1_5bf5, 0xa8c1_53f3, // ldp x27,x28 ... x19,x20, [sp], #16
1487:         0xf940_0000 | ((bpf_func / 8) << 10) | (14 << 5) | 16,   // ldr x16, [x14, #bpf_func]
1488:         add_imm(16, 16, entry),                                  // add x16, x16, #entry
1489:         0xd61f_0200,                                             // br x16
1490:     ];
1491:     let out = w.len() as u32;
1492:     for (at, base) in [(3usize, 0x5400_0002u32), (6, 0x5400_0002), (9, 0xb400_000e)] {
1493:         w[at] = base | ((out - at as u32) << 5);
1494:     }
1495:     Ok(w)
1496: }

631: static int emit_bpf_tail_call(struct jit_ctx *ctx)
632: {
633: 	/* bpf_tail_call(void *prog_ctx, struct bpf_array *array, u64 index) */
634: 	const u8 r2 = bpf2a64[BPF_REG_2];
635: 	const u8 r3 = bpf2a64[BPF_REG_3];
636: 
637: 	const u8 tmp = bpf2a64[TMP_REG_1];
638: 	const u8 prg = bpf2a64[TMP_REG_2];
639: 	const u8 tcc = bpf2a64[TMP_REG_3];
640: 	const u8 ptr = bpf2a64[TCCNT_PTR];
641: 	size_t off;
642: 	__le32 *branch1 = NULL;
643: 	__le32 *branch2 = NULL;
644: 	__le32 *branch3 = NULL;
645: 
646: 	/* if (index >= array->map.max_entries)
647: 	 *     goto out;
648: 	 */
649: 	off = offsetof(struct bpf_array, map.max_entries);
650: 	emit_a64_mov_i64(tmp, off, ctx);
651: 	emit(A64_LDR32(tmp, r2, tmp), ctx);
652: 	emit(A64_MOV(0, r3, r3), ctx);
653: 	emit(A64_CMP(0, r3, tmp), ctx);
654: 	branch1 = ctx->image + ctx->idx;
655: 	emit(A64_NOP, ctx);
656: 
657: 	/*
658: 	 * if ((*tail_call_cnt_ptr) >= MAX_TAIL_CALL_CNT)
659: 	 *     goto out;
660: 	 */
661: 	emit_a64_mov_i64(tmp, MAX_TAIL_CALL_CNT, ctx);
662: 	emit(A64_LDR64I(tcc, ptr, 0), ctx);
663: 	emit(A64_CMP(1, tcc, tmp), ctx);
664: 	branch2 = ctx->image + ctx->idx;
665: 	emit(A64_NOP, ctx);
666: 
667: 	/* (*tail_call_cnt_ptr)++; */
668: 	emit(A64_ADD_I(1, tcc, tcc, 1), ctx);
669: 
670: 	/* prog = array->ptrs[index];
671: 	 * if (prog == NULL)
672: 	 *     goto out;
673: 	 */
674: 	off = offsetof(struct bpf_array, ptrs);
675: 	emit_a64_mov_i64(tmp, off, ctx);
676: 	emit(A64_ADD(1, tmp, r2, tmp), ctx);
677: 	emit(A64_LSL(1, prg, r3, 3), ctx);
678: 	emit(A64_LDR64(prg, tmp, prg), ctx);
679: 	branch3 = ctx->image + ctx->idx;
680: 	emit(A64_NOP, ctx);
681: 
682: 	/* Update tail_call_cnt if the slot is populated. */
683: 	emit(A64_STR64I(tcc, ptr, 0), ctx);
684: 
685: 	/* restore SP */
686: 	if (ctx->stack_size && !ctx->priv_sp_used)
687: 		emit(A64_ADD_I(1, A64_SP, A64_SP, ctx->stack_size), ctx);
688: 
689: 	pop_callee_regs(ctx);
690: 
691: 	/* goto *(prog->bpf_func + prologue_offset); */
692: 	off = offsetof(struct bpf_prog, bpf_func);
693: 	emit_a64_mov_i64(tmp, off, ctx);
694: 	emit(A64_LDR64(tmp, prg, tmp), ctx);
695: 	emit(A64_ADD_I(1, tmp, tmp, sizeof(u32) * PROLOGUE_OFFSET), ctx);
696: 	emit(A64_BR(tmp), ctx);

701:
